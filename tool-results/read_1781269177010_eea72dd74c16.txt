    21→/**
    22→ * Compiler service that provides a four-step TRUSTED compilation pipeline:
    23→ *
    24→ * **Step 1 – Built-in (TypeScript)**: Fast, no-dependency syntax checking.
    25→ *
    26→ * **Step 2 – Transpiler (TypeScript)**: Converts .airo code to C++ Arduino/ESP32 code.
    27→ *
    28→ * **Step 3 – Python (airo_compiler)**: Full compilation pipeline using Python.
    29→ *    If Python is not installed, the TypeScript transpiler output is used.
    30→ *
    31→ * **Step 4 – Arduino CLI build**: Compiles the C++ into a .bin firmware file.
    32→ *    Auto-downloads Arduino CLI to ~/.airone/tools/ if not found in PATH.
    33→ *    Auto-installs ESP32 board support and required libraries.
    34→ *    Produces <sketchName>.ino.bin in the build directory.
    35→ */
    36→@injectable()
    37→export class AiroCompilerService {
    38→
    39→    @inject(AiroBuiltInCompiler)
    40→    protected readonly builtInCompiler!: AiroBuiltInCompiler;
    41→
    42→    @inject(AiroTranspiler)
    43→    protected readonly transpiler!: AiroTranspiler;
    44→
    45→    private pythonPath: string;
    46→    private compilerDir: string;
    47→    private cachedArduinoCli: string | undefined;
    48→    private _arduinoCliInstalling: boolean = false;
    49→
    50→    /** Directory where Airone stores tooling (Arduino CLI, esptool, etc.) */
    51→    private readonly toolsDir: string;
    52→
    53→    constructor() {
    54→        this.compilerDir = this.resolveCompilerDir();
    55→        this.pythonPath = this.resolvePythonPath();
    56→        this.toolsDir = path.join(os.homedir(), '.airone', 'tools');
    57→    }
    58→
    59→    private resolveCompilerDir(): string {
    60→        // In packaged app: resources/airo-compiler
    61→        if (typeof __dirname !== 'undefined' && __dirname.includes('.asar')) {
    62→            return path.join(process.resourcesPath!, 'airo-compiler');
    63→        }
    64→
    65→        // Dev mode - look for airo-compiler in common locations
    66→        const possibleLocations = [
    67→            path.resolve(__dirname, '../../../../../../airo-compiler'),
    68→            path.resolve(process.cwd(), 'airo-compiler'),
    69→            path.resolve(process.cwd(), '../airo-compiler'),
    70→        ];
    71→
    72→        try {
    73→            for (const loc of possibleLocations) {
    74→                if (fs.existsSync(path.join(loc, 'airo_compiler', '__init__.py'))) {
    75→                    return loc;
    76→                }
    77→            }
    78→        } catch {
    79→            // ignore
    80→        }
    81→
    82→        // Fallback
    83→        return possibleLocations[0];
    84→    }
    85→
    86→    private resolvePythonPath(): string {
    87→        return process.platform === 'win32' ? 'python' : 'python3';
    88→    }

    90→    /**
    91→     * Compile a .airo file through the full TRUSTED pipeline.
    92→     */
    93→    async compile(request: CompileRequest): Promise<CompileResult> {
    94→        // ─── Step 1: Built-in syntax check (always runs, no dependencies) ──
    95→        const builtInResult = await this.builtInCompiler.verify(request.filePath);
    96→
    97→        if (!builtInResult.success) {
    98→            return {
    99→                success: false,
   100→                output: builtInResult.output,
   101→                error: builtInResult.error || builtInResult.errors?.map(e => e.message).join('\n'),
   102→            };
   103→        }
   104→
   105→        // ─── Step 2: Transpile .airo → C++ (always runs) ──────────────
   106→        let airoCode: string;
   107→        const fsPath = request.filePath.startsWith('file://')
   108→            ? this.fsPathFromUri(request.filePath)
   109→            : request.filePath;
   110→
   111→        try {
   112→            airoCode = fs.readFileSync(fsPath, { encoding: 'utf8' });
   113→        } catch (err: unknown) {
   114→            return {
   115→                success: false,
   116→                output: '',
   117→                error: `Cannot read .airo file: ${err instanceof Error ? err.message : String(err)}`,
   118→            };
   119→        }
   120→
   121→        const sketchName = path.basename(fsPath, '.airo');
   122→        const transpileResult = this.transpiler.transpile(airoCode, sketchName);
   123→
   124→        // Write the generated C++ code to the output directory
   125→        const outputDir = request.outputDir || path.join(path.dirname(fsPath), 'build');
   126→        try {
   127→            if (!fs.existsSync(outputDir)) {
   128→                fs.mkdirSync(outputDir, { recursive: true });
   129→            }
   130→
   131→            // Arduino CLI requires the .ino file to be in a directory with the same name
   132→            // e.g., for sketch "titled", we need: build/titled/titled.ino
   133→            const sketchDir = path.join(outputDir, sketchName);
   134→            if (!fs.existsSync(sketchDir)) {
   135→                fs.mkdirSync(sketchDir, { recursive: true });
   136→            }
   137→
   138→            const cppPath = path.join(sketchDir, `${sketchName}.ino.cpp`);
   139→            fs.writeFileSync(cppPath, transpileResult.cppCode, { encoding: 'utf8' });
   140→
   141→            // Also write a minimal Arduino sketch .ino file
   142→            const inoPath = path.join(sketchDir, `${sketchName}.ino`);
   143→            if (!fs.existsSync(inoPath)) {
   144→                fs.writeFileSync(inoPath, `#include "${sketchName}.ino.cpp"\n`, { encoding: 'utf8' });
   145→            }
   146→
   147→            const generatedFiles = [cppPath, inoPath];
   148→            let combinedOutput = builtInResult.output + '\n' +
   149→                `✓ Step 2 — Transpiled to C++: ${cppPath}\n` +
   150→                `  Required libraries: ${transpileResult.requiredLibraries.join(', ') || 'none'}\n` +
   151→                (transpileResult.errors.length > 0 ? `  Warnings: ${transpileResult.errors.join('; ')}\n` : '');
   152→
   153→            // ─── Step 3: Python airo_compiler (always attempt) ────────
   154→            const pythonResult = await this.tryPythonCompile(request);
   155→
   156→            if (pythonResult) {
   157→                if (pythonResult.success) {
   158→                    combinedOutput += '\n✓ Step 3 — Python airo_compiler succeeded.\n' + pythonResult.output;
   159→                    if (pythonResult.generatedFiles) {
   160→                        generatedFiles.push(...pythonResult.generatedFiles);
   161→                    }
   162→                } else {
   163→                    combinedOutput +=
   164→                        '\n⚠ Step 3 — Python airo_compiler failed (using TypeScript transpiler output as fallback):\n' +
   165→                        `  ${pythonResult.error || pythonResult.output}\n` +
   166→                        '  The C++ file from Step 2 is ready for Arduino CLI.\n';
   167→                }
   168→            } else {
   169→                combinedOutput +=
   170→                    '\n⚠ Step 3 — Python airo_compiler not available.\n' +
   171→                    '  The TypeScript transpiler C++ output will be used (this is fine).\n';
   172→            }
   173→
   174→            // ─── Step 4: Arduino CLI build (auto-install if needed) ────
   175→            const fqbn = request.target === 'esp8266'
   176→                ? 'esp8266:esp8266:generic'
   177→                : 'esp32:esp32:esp32';
   178→
   179→            // Try to find or auto-install Arduino CLI
   180→            let arduinoCli = this.findArduinoCli();
   181→
   182→            if (!arduinoCli) {
   183→                combinedOutput += '\n⏳ Step 4 — Arduino CLI not found. Auto-installing...\n';
   184→                const installResult = await this.ensureArduinoCli(combinedOutput);
   185→                combinedOutput = installResult.output;
   186→
   187→                if (installResult.cliPath) {
   188→                    arduinoCli = installResult.cliPath;
   189→                } else {
   190→                    combinedOutput +=
   191→                        '\n⚠ Step 4 — Could not auto-install Arduino CLI.\n' +
   192→                        '  Firmware binary (.bin) not produced.\n' +
   193→                        '  Install manually: https://arduino.github.io/arduino-cli/latest/\n' +
   194→                        '  Then: arduino-cli core install esp32:esp32\n' +
   195→                        `  Then: arduino-cli compile --fqbn ${fqbn} "${outputDir}"\n`;
   196→
   197→                    return {
   198→                        success: true,
   199→                        output: combinedOutput,
   200→                        generatedFiles,
   201→                    };
   202→                }
   203→            }
   204→
   205→            // Ensure ESP32 board support is installed
   206→            const coreCheck = await this.ensureEsp32Core(arduinoCli, fqbn, combinedOutput);
   207→            combinedOutput = coreCheck.output;
   208→
   209→            // Install required libraries
   210→            if (transpileResult.requiredLibraries.length > 0) {
   211→                const libResult = await this.installRequiredLibs(
   212→                    arduinoCli, transpileResult.requiredLibraries, combinedOutput
   213→                );
   214→                combinedOutput = libResult.output;
   215→            }
   216→
   217→            // Now run the Arduino CLI build
   218→            const arduinoResult = await this.tryArduinoBuildWithCli(arduinoCli, outputDir, sketchName, fqbn);
   219→
   220→            if (arduinoResult.success) {
   221→                combinedOutput += '\n✓ Step 4 — Arduino CLI build succeeded.\n' + arduinoResult.output;
   222→                if (arduinoResult.generatedFiles) {
   223→                    generatedFiles.push(...arduinoResult.generatedFiles);
   224→                }
   225→                return {
   226→                    success: true,
   227→                    output: combinedOutput,
   228→                    generatedFiles,
   229→                    binaryPath: arduinoResult.binaryPath,
   230→                };
   231→            }
   232→
   233→            // Arduino CLI ran but build failed
   234→            combinedOutput +=
   235→                '\n✗ Step 4 — Arduino CLI build failed:\n' +
   236→                `  ${arduinoResult.error || arduinoResult.output}\n`;
   237→
   238→            return {
   239→                success: true, // C++ was generated even if Arduino build failed
   240→                output: combinedOutput,
   241→                generatedFiles,
   242→            };
   243→        } catch (err: unknown) {
   244→            const message = err instanceof Error ? err.message : String(err);
   245→            return {
   246→                success: false,
   247→                output: '',
   248→                error: `Failed to write compiled output: ${message}`,
   249→            };
   250→        }
   251→    }

   254→     * Verify using the built-in TypeScript compiler (fast, no dependencies).

   256→    async verifyBuiltIn(filePath: string): Promise<import('../common/airo-protocol').VerifyResult> {

   260→    /**
   261→     * Check if arduino-cli is available — first in PATH, then in ~/.airone/tools/.
   262→     * Returns the command/path string if found, or undefined.
   263→     */
   264→    findArduinoCli(): string | undefined {
   265→        if (this.cachedArduinoCli !== undefined) {
   266→            return this.cachedArduinoCli;
   267→        }
   268→
   269→        // 1. Check system PATH
   270→        try {
   271→            const isWin = process.platform === 'win32';
   272→            const checkCmd = isWin ? 'where' : 'which';
   273→            execSync(`${checkCmd} arduino-cli`, { stdio: 'ignore', timeout: 5000 });
   274→            this.cachedArduinoCli = 'arduino-cli';
   275→            return this.cachedArduinoCli;
   276→        } catch {
   277→            // not in PATH
   278→        }
   279→
   280→        // 2. Check local tools directory
   281→        const localCliPath = this.getLocalArduinoCliPath();
   282→        if (localCliPath && fs.existsSync(localCliPath)) {
   283→            this.cachedArduinoCli = localCliPath;
   284→            return this.cachedArduinoCli;
   285→        }
   286→
   287→        this.cachedArduinoCli = undefined;
   288→        return undefined;
   289→    }
   290→
   291→    /**
   292→     * Get the expected path for a locally-installed Arduino CLI.
   293→     */
   294→    private getLocalArduinoCliPath(): string {
   295→        const isWin = process.platform === 'win32';
   296→        return path.join(this.toolsDir, isWin ? 'arduino-cli.exe' : 'arduino-cli');
   297→    }

   299→    /**
   300→     * Auto-download and install Arduino CLI to ~/.airone/tools/.
   301→     *
   302→     * Downloads the latest stable release from GitHub releases API.
   303→     * Returns the path to the installed binary, or undefined on failure.
   304→     */
   305→    async ensureArduinoCli(currentOutput: string): Promise<{ cliPath: string | undefined; output: string }> {
   306→        let output = currentOutput;
   307→
   308→        // Prevent concurrent installations
   309→        if (this._arduinoCliInstalling) {
   310→            output += '  Arduino CLI installation already in progress...\n';
   311→            return { cliPath: undefined, output };
   312→        }
   313→
   314→        // Check if already installed locally
   315→        const localPath = this.getLocalArduinoCliPath();
   316→        if (fs.existsSync(localPath)) {
   317→            this.cachedArduinoCli = localPath;
   318→            output += `  ✓ Found local Arduino CLI: ${localPath}\n`;
   319→            return { cliPath: localPath, output };
   320→        }
   321→
   322→        this._arduinoCliInstalling = true;
   323→
   324→        try {
   325→            // Create tools directory
   326→            if (!fs.existsSync(this.toolsDir)) {
   327→                fs.mkdirSync(this.toolsDir, { recursive: true });
   328→            }
   329→
   330→            // Determine platform-specific download URL
   331→            const platform = this.getArduinoCliPlatform();
   332→            const isWindows = platform.toLowerCase().includes('windows');
   333→            const ext = isWindows ? '.zip' : '.tar.gz';
   334→
   335→            // Step 1: Get the latest version tag from GitHub releases API
   336→            output += '  Fetching latest Arduino CLI version from GitHub...\n';
   337→            let version: string | undefined;
   338→            try {
   339→                version = await this.getLatestArduinoCliVersion();
   340→                output += `  Latest version: ${version}\n`;
   341→            } catch {
   342→                output += '  ⚠ Could not fetch version from GitHub API, trying direct download...\n';
   343→            }
   344→
   345→            // Construct download URL — prefer GitHub releases with version
   346→            // NOTE: The tag path uses the version WITH 'v' (e.g. v1.5.0),
   347→            // but the asset FILENAME uses version WITHOUT 'v' (e.g. 1.5.0).
   348→            // Actual URL: https://github.com/arduino/arduino-cli/releases/download/v1.5.0/arduino-cli_1.5.0_Windows_64bit.zip
   349→            let downloadUrl: string;
   350→            if (version) {
   351→                const versionNum = version.replace(/^v/, ''); // Strip 'v' prefix for filename
   352→                downloadUrl = `https://github.com/arduino/arduino-cli/releases/download/${version}/arduino-cli_${versionNum}_${platform}${ext}`;
   353→            } else {
   354→                // Fallback: try downloads.arduino.cc with .zip for Windows
   355→                downloadUrl = `https://downloads.arduino.cc/arduino-cli/arduino-cli_latest_${platform}${ext}`;
   356→            }
   357→
   358→            const fileName = `arduino-cli_${platform}${ext}`;
   359→
   360→            output += `  Downloading Arduino CLI for ${platform}...\n`;
   361→            output += `  URL: ${downloadUrl}\n`;
   362→
   363→            // Download the archive
   364→            const archivePath = path.join(this.toolsDir, fileName);
   365→            await this.downloadFile(downloadUrl, archivePath);
   366→
   367→            output += '  ✓ Download complete. Extracting...\n';
   368→
   369→            // Extract the binary
   370→            const extractedPath = await this.extractArchive(archivePath, localPath, isWindows);
   371→
   372→            if (extractedPath && fs.existsSync(extractedPath)) {
   373→                // Make executable on Unix
   374→                if (process.platform !== 'win32') {
   375→                    try {
   376→                        fs.chmodSync(extractedPath, 0o755);
   377→                    } catch { /* ignore */ }
   378→                }
   379→
   380→                // Verify it works
   381→                try {
   382→                    const version = execSync(`"${extractedPath}" version`, { encoding: 'utf8', timeout: 10000 });
   383→                    output += `  ✓ Arduino CLI installed: ${version.trim()}\n`;
   384→                    output += `  Location: ${extractedPath}\n`;
   385→                } catch {
   386→                    output += `  ✓ Arduino CLI installed at: ${extractedPath}\n`;
   387→                }
   388→
   389→                this.cachedArduinoCli = extractedPath;
   390→
   391→                // Clean up archive
   392→                try { fs.unlinkSync(archivePath); } catch { /* ignore */ }
   393→
   394→                return { cliPath: extractedPath, output };
   395→            }
   396→
   397→            output += '  ✗ Failed to extract Arduino CLI.\n';
   398→            return { cliPath: undefined, output };
   399→
   400→        } catch (err: unknown) {
   401→            const message = err instanceof Error ? err.message : String(err);
   402→            output += `  ✗ Arduino CLI auto-install failed: ${message}\n`;
   403→            return { cliPath: undefined, output };
   404→        } finally {
   405→            this._arduinoCliInstalling = false;
   406→        }
   407→    }

   409→    /**
   410→     * Ensure ESP32 core is installed for the given FQBN.
   411→     */
   412→    async ensureEsp32Core(
   413→        arduinoCli: string,
   414→        fqbn: string,
   415→        currentOutput: string
   416→    ): Promise<{ output: string }> {
   417→        let output = currentOutput;
   418→
   419→        try {
   420→            // Check if the core is already installed
   421→            // IMPORTANT: Must use --config-dir to match the directory where
   422→            // we install cores. Without this, arduino-cli looks in the default
   423→            // location and won't find our ESP32 core.
   424→            const configDir = path.join(os.homedir(), '.airone', 'arduino-cli');
   425→            const coreListArgs = fs.existsSync(configDir)
   426→                ? `"${arduinoCli}" core list --config-dir "${configDir}"`
   427→                : `"${arduinoCli}" core list`;
   428→            const coreList = execSync(coreListArgs, {
   429→                encoding: 'utf8',
   430→                timeout: 15000,
   431→            });
   432→
   433→            if (coreList.includes('esp32:esp32') || coreList.includes('esp8266:esp8266')) {
   434→                output += '  ✓ ESP32 board support already installed.\n';
   435→                return { output };
   436→            }
   437→        } catch {
   438→            // core list might fail, try installing anyway
   439→        }
   440→
   441→        output += '  ⏳ Installing ESP32 board support (first time, this may take a few minutes)...\n';
   442→
   443→        try {
   444→            // Add ESP32 board URL
   445→            const configDir = path.join(os.homedir(), '.airone', 'arduino-cli');
   446→            if (!fs.existsSync(configDir)) {
   447→                fs.mkdirSync(configDir, { recursive: true });
   448→            }
   449→
   450→            // Initialize config if needed
   451→            try {
   452→                execSync(`"${arduinoCli}" config init --dest-dir "${configDir}"`, {
   453→                    encoding: 'utf8',
   454→                    timeout: 10000,
   455→                });
   456→            } catch { /* may already exist */ }
   457→
   458→            // Set data directory to our custom location so cores/packages
   459→            // are stored alongside our config (not in the default ~/.arduino15)
   460→            try {
   461→                execSync(`"${arduinoCli}" config set directories.data "${path.join(configDir, 'data')}" --dest-dir "${configDir}"`, {
   462→                    encoding: 'utf8',
   463→                    timeout: 10000,
   464→                });
   465→            } catch { /* may already be set */ }
   466→            try {
   467→                execSync(`"${arduinoCli}" config set directories.downloads "${path.join(configDir, 'staging')}" --dest-dir "${configDir}"`, {
   468→                    encoding: 'utf8',
   469→                    timeout: 10000,
   470→                });
   471→            } catch { /* may already be set */ }
   472→
   473→            // Add ESP32 board index URL
   474→            const esp32Url = 'https://espressif.github.io/arduino-esp32/package_esp32_index.json';
   475→            try {
   476→                execSync(`"${arduinoCli}" config add board_manager.additional_urls ${esp32Url} --dest-dir "${configDir}"`, {
   477→                    encoding: 'utf8',
   478→                    timeout: 10000,
   479→                });
   480→            } catch { /* may already be added */ }
   481→
   482→            // Update index
   483→            output += '  Updating board index...\n';
   484→            try {
   485→                execSync(`"${arduinoCli}" core update-index --config-dir "${configDir}"`, {
   486→                    encoding: 'utf8',
   487→                    timeout: 60000,
   488→                });
   489→            } catch { /* ignore errors from update-index */ }
   490→
   491→            // Install ESP32 core
   492→            const coreName = fqbn.split(':').slice(0, 2).join(':'); // e.g. esp32:esp32
   493→            output += `  Installing core: ${coreName}...\n`;
   494→
   495→            try {
   496→                const installOutput = execSync(
   497→                    `"${arduinoCli}" core install ${coreName} --config-dir "${configDir}"`,
   498→                    { encoding: 'utf8', timeout: 300_000 } // 5 min timeout
   499→                );
   500→                output += '  ✓ ESP32 board support installed successfully.\n';
   501→                return { output };
   502→            } catch (installErr: unknown) {
   503→                const installMsg = installErr instanceof Error ? installErr.message : String(installErr);
   504→
   505→                // If the error is about a corrupted archive or locked file, try cleaning
   506→                // staging directories and retrying once.
   507→                if (installMsg.includes('corrupted') || installMsg.includes('locked') ||
   508→                    installMsg.includes('being used by another process') || installMsg.includes('EBUSY')) {
   509→                    output += `  ⚠ Install failed (locked/corrupted file). Cleaning staging files and retrying...\n`;
   510→
   511→                    // Clean staging directories in the Arduino data directory
   512→                    const stagingDir = path.join(configDir, 'staging');
   513→                    const packagesDir = path.join(os.homedir(), '.airone', 'arduino-cli', 'packages');
   514→
   515→                    for (const dir of [stagingDir, packagesDir]) {
   516→                        try {
   517→                            if (fs.existsSync(dir)) {
   518→                                fs.rmSync(dir, { recursive: true, force: true });
   519→                                output += `  Cleaned: ${dir}\n`;
   520→                            }
   521→                        } catch (cleanErr) {
   522→                            output += `  Could not clean ${dir}: ${cleanErr instanceof Error ? cleanErr.message : String(cleanErr)}\n`;
   523→                        }
   524→                    }
   525→
   526→                    // Retry the install
   527→                    try {
   528→                        const retryOutput = execSync(
   529→                            `"${arduinoCli}" core install ${coreName} --config-dir "${configDir}"`,
   530→                            { encoding: 'utf8', timeout: 300_000 }
   531→                        );
   532→                        output += '  ✓ ESP32 board support installed successfully (after cleanup retry).\n';
   533→                        return { output };
   534→                    } catch (retryErr: unknown) {
   535→                        const retryMsg = retryErr instanceof Error ? retryErr.message : String(retryErr);
   536→                        output += `  ⚠ Retry also failed: ${retryMsg}\n`;
   537→                    }
   538→                }
   539→
   540→                output += `  ⚠ Could not auto-install ESP32 board support: ${installMsg}\n`;
   541→                output += '  You may need to install it manually: arduino-cli core install esp32:esp32\n';
   542→                return { output };
   543→            }
   544→        } catch (err: unknown) {
   545→            const message = err instanceof Error ? err.message : String(err);
   546→            output += `  ⚠ Could not auto-install ESP32 board support: ${message}\n`;
   547→            output += '  You may need to install it manually: arduino-cli core install esp32:esp32\n';
   548→            return { output };
   549→        }
   550→    }

   610→    /**
   611→     * Run the actual Arduino CLI build with a known CLI path.
   612→     */
   613→    private async tryArduinoBuildWithCli(
   614→        arduinoCli: string,
   615→        outputDir: string,
   616→        sketchName: string,
   617→        fqbn: string
   618→    ): Promise<CompileResult> {
   619→        const configDir = path.join(os.homedir(), '.airone', 'arduino-cli');
   620→
   621→        return new Promise(resolve => {
   622→            const args = [
   623→                'compile',
   624→                '--fqbn', fqbn,
   625→                '--output-dir', outputDir,
   626→            ];
   627→
   628→            // Use our config directory if it exists
   629→            if (fs.existsSync(configDir)) {
   630→                args.push('--config-dir', configDir);
   631→            }
   632→
   633→            // Arduino CLI requires the sketch directory (containing the .ino file)
   634→            // to match the .ino filename — use the sketch subdirectory, not the build root
   635→            const sketchDir = path.join(outputDir, sketchName);
   636→            if (fs.existsSync(sketchDir)) {
   637→                args.push(sketchDir);
   638→            } else {
   639→                args.push(outputDir);
   640→            }
   641→
   642→            const proc = spawn(arduinoCli, args, {
   643→                stdio: 'pipe',
   644→            });
   645→
   646→            let stdout = '';
   647→            let stderr = '';
   648→
   649→            proc.stdout.on('data', (data: Buffer) => {
   650→                stdout += data.toString();
   651→            });
   652→
   653→            proc.stderr.on('data', (data: Buffer) => {
   654→                stderr += data.toString();
   655→            });
   656→
   657→            proc.on('close', (code: number | null) => {
   658→                const expectedBinPath = path.join(outputDir, `${sketchName}.ino.bin`);
   659→                const binExists = fs.existsSync(expectedBinPath);
   660→
   661→                if (code === 0 && binExists) {
   662→                    resolve({
   663→                        success: true,
   664→                        output: `  Firmware binary: ${expectedBinPath}`,
   665→                        generatedFiles: [expectedBinPath],
   666→                        binaryPath: expectedBinPath,
   667→                    });
   668→                } else if (code === 0 && !binExists) {
   669→                    const altBinPath = this.findBinaryInDir(outputDir, sketchName);
   670→                    if (altBinPath) {
   671→                        resolve({
   672→                            success: true,
   673→                            output: `  Firmware binary: ${altBinPath}`,
   674→                            generatedFiles: [altBinPath],
   675→                            binaryPath: altBinPath,
   676→                        });
   677→                    } else {
   678→                        resolve({
   679→                            success: false,
   680→                            output: stdout,
   681→                            error: 'Arduino CLI reported success but no .bin file was found.',
   682→                        });
   683→                    }
   684→                } else {
   685→                    resolve({
   686→                        success: false,
   687→                        output: stdout,
   688→                        error: stderr || `Arduino CLI exited with code ${code}`,
   689→                    });
   690→                }
   691→            });
   692→
   693→            proc.on('error', () => {
   694→                resolve({
   695→                    success: false,
   696→                    output: '',
   697→                    error: 'Failed to start Arduino CLI.',
   698→                });
   699→            });
   700→
   701→            // 180 second timeout — first builds can be very slow (compiling ESP32 core)
   702→            setTimeout(() => {
   703→                proc.kill();
   704→                resolve({
   705→                    success: false,
   706→                    output: stdout,
   707→                    error: 'Arduino CLI build timed out after 180 seconds.',
   708→                });
   709→            }, 180_000);
   710→        });
   711→    }

   867→    /**
   868→     * Fetch the latest Arduino CLI version tag from the GitHub releases API.
   869→     * Returns the tag name (e.g. "v1.0.4") or throws on failure.
   870→     */
   871→    private getLatestArduinoCliVersion(): Promise<string> {
   872→        return new Promise((resolve, reject) => {
   873→            const options: https.RequestOptions = {
   874→                hostname: 'api.github.com',
   875→                path: '/repos/arduino/arduino-cli/releases/latest',
   876→                method: 'GET',
   877→                headers: {
   878→                    'User-Agent': 'Airone-IDE/1.0',
   879→                    'Accept': 'application/vnd.github+json',
   880→                },
   881→            };
   882→
   883→            const req = https.request(options, (res) => {
   884→                if (res.statusCode && res.statusCode >= 300 && res.statusCode < 400 && res.headers.location) {
   885→                    // Follow redirect
   886→                    const redirectUrl = new URL(res.headers.location);
   887→                    this.getLatestArduinoCliVersion().then(resolve).catch(reject);
   888→                    return;
   889→                }
   890→
   891→                if (res.statusCode !== 200) {
   892→                    reject(new Error(`GitHub API returned status ${res.statusCode}`));
   893→                    return;
   894→                }
   895→
   896→                let data = '';
   897→                res.on('data', (chunk: Buffer) => { data += chunk.toString(); });
   898→                res.on('end', () => {
   899→                    try {
   900→                        const release = JSON.parse(data);
   901→                        const tag = release.tag_name;
   902→                        if (tag) {
   903→                            resolve(tag);
   904→                        } else {
   905→                            reject(new Error('No tag_name in GitHub release response'));
   906→                        }
   907→                    } catch {
   908→                        reject(new Error('Failed to parse GitHub API response'));
   909→                    }
   910→                });
   911→            });
   912→
   913→            req.on('error', reject);
   914→            req.setTimeout(15000, () => {
   915→                req.destroy();
   916→                reject(new Error('GitHub API request timed out'));
   917→            });
   918→            req.end();
   919→        });
   920→    }

   922→    /**
   923→     * Download a file from a URL with redirect support.
   924→     * Handles both HTTP and HTTPS redirects (GitHub releases may redirect
   925→     * to objects.githubusercontent.com which is HTTPS, but some CDNs use HTTP).
   926→     */
   927→    private downloadFile(url: string, destPath: string): Promise<void> {
   928→        return new Promise((resolve, reject) => {
   929→            const doDownload = (currentUrl: string, redirectCount = 0) => {
   930→                if (redirectCount > 5) {
   931→                    reject(new Error('Too many redirects'));
   932→                    return;
   933→                }
   934→
   935→                const parsedUrl = new URL(currentUrl);
   936→                const isHttps = parsedUrl.protocol === 'https:';
   937→                const httpModule = isHttps ? https : http;
   938→                const options: https.RequestOptions = {
   939→                    hostname: parsedUrl.hostname,
   940→                    path: parsedUrl.pathname + parsedUrl.search,
   941→                    method: 'GET',
   942→                    headers: { 'User-Agent': 'Airone-IDE/1.0' },
   943→                };
   944→
   945→                const req = httpModule.request(options, (res) => {
   946→                    // Handle redirects
   947→                    if (res.statusCode && res.statusCode >= 300 && res.statusCode < 400 && res.headers.location) {
   948→                        doDownload(res.headers.location, redirectCount + 1);
   949→                        return;
   950→                    }
   951→
   952→                    if (res.statusCode !== 200) {
   953→                        reject(new Error(`Download failed with status ${res.statusCode}`));
   954→                        return;
   955→                    }
   956→
   957→                    const file = fs.createWriteStream(destPath);
   958→                    res.pipe(file);
   959→
   960→                    file.on('finish', () => {
   961→                        file.close();
   962→                        resolve();
   963→                    });
   964→
   965→                    file.on('error', (err) => {
   966→                        fs.unlinkSync(destPath);
   967→                        reject(err);
   968→                    });
   969→                });
   970→
   971→                req.on('error', reject);
   972→                req.setTimeout(120_000, () => {
   973→                    req.destroy();
   974→                    reject(new Error('Download timed out'));
   975→                });
   976→                req.end();
   977→            };
   978→
   979→            doDownload(url);
   980→        });
   981→    }

   983→    /**
   984→     * Extract Arduino CLI binary from a downloaded archive.
   985→     */
   986→    private async extractArchive(archivePath: string, targetBinaryPath: string, isWindows: boolean): Promise<string | undefined> {
   987→        try {
   988→            const extractDir = path.dirname(archivePath);
   989→
   990→            if (isWindows) {
   991→                // Extract .zip on Windows — try multiple methods for robustness
   992→                let extracted = false;
   993→
   994→                // Method 1: PowerShell Expand-Archive
   995→                // Use -EncodedCommand to avoid all quoting/escaping issues with
   996→                // paths that contain spaces, parentheses, or special characters.
   997→                if (!extracted) {
   998→                    try {
   999→                        const psScript = `Expand-Archive -Path '${archivePath.replace(/'/g, "''")}' -DestinationPath '${extractDir.replace(/'/g, "''")}' -Force`;
  1000→                        const encodedCmd = Buffer.from(psScript).toString('base64');
  1001→                        execSync(`powershell -NoProfile -EncodedCommand ${encodedCmd}`, { timeout: 60000 });
  1002→                        extracted = true;
  1003→                    } catch (psErr) {
  1004→                        // Fallback: try the old quoting approach (for older PowerShell)
  1005→                        try {
  1006→                            const safeArchivePath = archivePath.replace(/'/g, "''");
  1007→                            const safeExtractDir = extractDir.replace(/'/g, "''");
  1008→                            const psCmd = `Expand-Archive -Path '${safeArchivePath}' -DestinationPath '${safeExtractDir}' -Force`;
  1009→                            execSync(`powershell -NoProfile -Command "${psCmd}"`, { timeout: 60000 });
  1010→                            extracted = true;
  1011→                        } catch (psErr2) {
  1012→                            console.warn('[AiroCompilerService] PowerShell extraction failed, trying fallback:', psErr2 instanceof Error ? psErr2.message : String(psErr2));
  1013→                        }
  1014→                    }
  1015→                }
  1016→
  1017→                // Method 2: Use Node.js built-in zlib + unzip via tar (Windows 10+ has tar)
  1018→                if (!extracted) {
  1019→                    try {
  1020→                        execSync(`tar -xf "${archivePath}" -C "${extractDir}"`, { timeout: 60000 });
  1021→                        extracted = true;
  1022→                    } catch (tarErr) {
  1023→                        console.warn('[AiroCompilerService] tar extraction failed:', tarErr instanceof Error ? tarErr.message : String(tarErr));
  1024→                    }
  1025→                }
  1026→
  1027→                // Method 3: Manual ZIP extraction using Node.js (no external tools)
  1028→                if (!extracted) {
  1029→                    try {
  1030→                        extracted = await this.extractZipManually(archivePath, extractDir);
  1031→                    } catch (manualErr) {
  1032→                        console.warn('[AiroCompilerService] Manual ZIP extraction failed:', manualErr instanceof Error ? manualErr.message : String(manualErr));
  1033→                    }
  1034→                }
  1035→
  1036→                if (!extracted) {
  1037→                    console.error('[AiroCompilerService] All extraction methods failed.');
  1038→                    return undefined;
  1039→                }
  1040→
  1041→                // Find the extracted binary — search multiple possible locations
  1042→                const possiblePaths = [
  1043→                    targetBinaryPath,                         // If already in the right place
  1044→                    path.join(extractDir, 'arduino-cli.exe'),
  1045→                    path.join(extractDir, 'bin', 'arduino-cli.exe'),
  1046→                    // The zip may extract to a subdirectory like arduino-cli_1.5.1_Windows_64bit/
  1047→                ];
  1048→
  1049→                for (const p of possiblePaths) {
  1050→                    if (fs.existsSync(p)) {
  1051→                        if (p !== targetBinaryPath) {
  1052→                            fs.copyFileSync(p, targetBinaryPath);
  1053→                        }
  1054→                        return targetBinaryPath;
  1055→                    }
  1056→                }
  1057→
  1058→                // Search recursively in the extraction directory
  1059→                const found = this.findFileRecursive(extractDir, 'arduino-cli.exe', targetBinaryPath);
  1060→                if (found) {
  1061→                    return found;
  1062→                }
  1063→
  1064→                console.error('[AiroCompilerService] Could not find arduino-cli.exe after extraction.');
  1065→                return undefined;
  1066→            } else {
  1067→                // Extract .tar.gz on Unix
  1068→                const extractDir = path.dirname(archivePath);
  1069→                execSync(`tar -xzf "${archivePath}" -C "${extractDir}"`, { timeout: 30000 });
  1070→
  1071→                const possiblePaths = [
  1072→                    path.join(extractDir, 'arduino-cli'),
  1073→                    path.join(extractDir, 'bin', 'arduino-cli'),
  1074→                ];
  1075→
  1076→                for (const p of possiblePaths) {
  1077→                    if (fs.existsSync(p)) {
  1078→                        if (p !== targetBinaryPath) {
  1079→                            fs.copyFileSync(p, targetBinaryPath);
  1080→                        }
  1081→                        try { fs.chmodSync(targetBinaryPath, 0o755); } catch { /* ignore */ }
  1082→                        return targetBinaryPath;
  1083→                    }
  1084→                }
  1085→
  1086→                // Search recursively
  1087→                return this.findFileRecursive(extractDir, 'arduino-cli', targetBinaryPath);
  1088→            }
  1089→        } catch (err: unknown) {
  1090→            const message = err instanceof Error ? err.message : String(err);
  1091→            console.error('[AiroCompilerService] Archive extraction failed:', message);
  1092→            return undefined;
  1093→        }
  1094→    }

  1096→    /**
  1097→     * Manually extract a ZIP file using .NET ZipFile via PowerShell.
  1098→     * Fallback when Expand-Archive and tar are not available on Windows.
  1099→     */
  1100→    private async extractZipManually(archivePath: string, destDir: string): Promise<boolean> {
  1101→        return new Promise((resolve) => {
  1102→            try {
  1103→                // Replace backslashes with forward slashes for PowerShell compatibility
  1104→                const safeArchive = archivePath.replace(/\\/g, '/');
  1105→                const safeDest = destDir.replace(/\\/g, '/');
  1106→
  1107→                // Use .NET's ZipFile via PowerShell (more reliable than Expand-Archive)
  1108→                const psCmd =
  1109→                    `Add-Type -AssemblyName System.IO.Compression.FileSystem; ` +
  1110→                    `[System.IO.Compression.ZipFile]::ExtractToDirectory('${safeArchive}', '${safeDest}', $true)`;
  1111→
  1112→                execSync(`powershell -NoProfile -Command "${psCmd}"`, { timeout: 60000 });
  1113→                resolve(true);
  1114→            } catch (err) {
  1115→                console.warn('[AiroCompilerService] .NET ZipFile extraction also failed:', err instanceof Error ? err.message : String(err));
  1116→                resolve(false);
  1117→            }
  1118→        });
  1119→    }

  1121→    /**
  1122→     * Search recursively for a file and copy it to target.
  1123→     */
  1124→    private findFileRecursive(dir: string, fileName: string, targetPath: string): string | undefined {
  1125→        try {
  1126→            const entries = fs.readdirSync(dir, { withFileTypes: true });
  1127→            for (const entry of entries) {
  1128→                const fullPath = path.join(dir, entry.name);
  1129→                if (entry.isDirectory()) {
  1130→                    const result = this.findFileRecursive(fullPath, fileName, targetPath);
  1131→                    if (result) return result;
  1132→                } else if (entry.name === fileName) {
  1133→                    fs.copyFileSync(fullPath, targetPath);
  1134→                    try { fs.chmodSync(targetPath, 0o755); } catch { /* ignore */ }
  1135→                    return targetPath;
  1136→                }
  1137→            }
  1138→        } catch { /* ignore */ }
  1139→        return undefined;
  1140→    }
