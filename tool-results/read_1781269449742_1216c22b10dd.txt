   400→   400→   400→                };
   401→   401→   401→            }
   402→   402→   402→            portPath = detected.path;
   403→   403→   403→        }
   404→   404→   404→
   405→   405→   405→        // ── Locate esptool ────────────────────────────────────────────────
   406→   406→   406→        const esptoolCmd = await this.findEsptool();
   407→   407→   407→        if (!esptoolCmd) {
   408→   408→   408→            return {
   409→   409→   409→                success: false,
   410→   410→   410→                output: '',
   411→   411→   411→                error: 'esptool.py is not installed. Install it with: pip install esptool',
   412→   412→   412→            };
   413→   413→   413→        }
   414→   414→   414→
   415→   415→   415→        // ── Resolve chip type & flash offset ──────────────────────────────
   416→   416→   416→        const chipType = this.normalizeChipType(request.chipType);
   417→   417→   417→        if (!chipType) {
   418→   418→   418→            return {
   419→   419→   419→                success: false,
   420→   420→   420→                output: '',
   421→   421→   421→                error: `Unsupported chip type: "${request.chipType}". Supported: esp32, esp32s2, esp32s3, esp32c3, esp8266`,
   422→   422→   422→            };
   423→   423→   423→        }
   424→   424→   424→
   425→   425→   425→        const flashOffset = request.flashOffset || CHIP_FLASH_OFFSETS[chipType] || '0x10000';
   426→   426→   426→        const baudRate = request.baudRate || DEFAULT_BAUD_RATE;
   427→   427→   427→
   428→   428→   428→        // ── Build command ─────────────────────────────────────────────────
   429→   429→   429→        const useModule = esptoolCmd.includes('-m esptool');
   430→   430→   430→        let cmd: string;
   431→   431→   431→        let args: string[];
   432→   432→   432→
   433→   433→   433→        if (useModule) {
   434→   434→   434→            // e.g. "python3 -m esptool" → cmd="python3", args=["-m","esptool",...]
   435→   435→   435→            const parts = esptoolCmd.split(' ');
   436→   436→   436→            cmd = parts[0];
   437→   437→   437→            args = [
   438→   438→   438→                ...parts.slice(1),
   439→   439→   439→                '--chip', chipType,
   440→   440→   440→                '--port', portPath,
   441→   441→   441→                '--baud', String(baudRate),
   442→   442→   442→                'write_flash',
   443→   443→   443→                '-z',
   444→   444→   444→                flashOffset,
   445→   445→   445→                request.binaryPath,
   446→   446→   446→            ];
   447→   447→   447→        } else {
   448→   448→   448→            // e.g. "esptool.py" or "esptool"
   449→   449→   449→            cmd = esptoolCmd;
   450→   450→   450→            args = [
   451→   451→   451→                '--chip', chipType,
   452→   452→   452→                '--port', portPath,
   453→   453→   453→                '--baud', String(baudRate),
   454→   454→   454→                'write_flash',
   455→   455→   455→                '-z',
   456→   456→   456→                flashOffset,
   457→   457→   457→                request.binaryPath,
   458→   458→   458→            ];
   459→   459→   459→        }
   460→   460→   460→
   461→   461→   461→        // ── Execute ───────────────────────────────────────────────────────
   462→   462→   462→        return this.executeFlash(cmd, args, portPath, onProgress);
   463→   463→   463→    }

   465→   465→   465→    // ─── Full Pipeline: Compile + Flash ──────────────────────────────────
   466→   466→   466→
   467→   467→   467→    /**
   468→   468→   468→     * Compile an .airo file and flash the resulting firmware to an ESP32 board.
   469→   469→   469→     *
   470→   470→   470→     * This method handles the full TRUSTED pipeline on the backend:
   471→   471→   471→     * 1. Compile the .airo file → C++ (via AiroCompilerService)
   472→   472→   472→     * 2. Build C++ → .bin (via Arduino CLI, if available)
   473→   473→   473→     * 3. Flash .bin → ESP32 (via esptool)
   474→   474→   474→     *
   475→   475→   475→     * @param airoFilePath  Absolute path to the .airo file (or file:// URI)
   476→   476→   476→     * @param chipType      Target chip: esp32, esp32s2, esp32s3, esp32c3, esp8266
   477→   477→   477→     * @param portPath      Optional serial port path. Auto-detected if omitted.
   478→   478→   478→     */
   479→   479→   479→    async flashAiroFile(airoFilePath: string, chipType: string, portPath?: string): Promise<FlashResult> {
   480→   480→   480→        // ── Resolve filesystem path ───────────────────────────────────
   481→   481→   481→        const fsPath = airoFilePath.startsWith('file://')
   482→   482→   482→            ? this.fsPathFromUri(airoFilePath)
   483→   483→   483→            : airoFilePath;
   484→   484→   484→
   485→   485→   485→        if (!fs.existsSync(fsPath)) {
   486→   486→   486→            return {
   487→   487→   487→                success: false,
   488→   488→   488→                output: '',
   489→   489→   489→                error: `File not found: ${fsPath}`,
   490→   490→   490→            };
   491→   491→   491→        }
   492→   492→   492→
   493→   493→   493→        const sketchName = path.basename(fsPath, '.airo');
   494→   494→   494→        const buildDir = path.join(path.dirname(fsPath), 'build');
   495→   495→   495→
   496→   496→   496→        // ── Step 1: Compile .airo → C++ → .bin ────────────────────────
   497→   497→   497→        const fqbn = chipType === 'esp8266'
   498→   498→   498→            ? 'esp8266:esp8266:generic'
   499→   499→   499→            : 'esp32:esp32:esp32';
   500→   500→   500→
   501→   501→   501→        const compileResult = await this.compilerService.compile({
   502→   502→   502→            filePath: fsPath,
   503→   503→   503→            target: chipType,
   504→   504→   504→            outputDir: buildDir,
   505→   505→   505→        });
   506→   506→   506→
   507→   507→   507→        if (!compileResult.success) {
   508→   508→   508→            return {
   509→   509→   509→                success: false,
   510→   510→   510→                output: compileResult.output,
   511→   511→   511→                error: compileResult.error || 'Compilation failed',
   512→   512→   512→            };
   513→   513→   513→        }
   514→   514→   514→
   515→   515→   515→        // ── Step 2: Locate the .bin firmware ──────────────────────────
   516→   516→   516→        let binaryPath = compileResult.binaryPath;
   517→   517→   517→
   518→   518→   518→        if (!binaryPath) {
   519→   519→   519→            // The compile result didn't include binaryPath — check the build directory
   520→   520→   520→            const expectedBin = path.join(buildDir, `${sketchName}.ino.bin`);
   521→   521→   521→            if (fs.existsSync(expectedBin)) {
   522→   522→   522→                binaryPath = expectedBin;
   523→   523→   523→            } else {
   524→   524→   524→                // Try Arduino CLI build directly if it wasn't attempted during compile
   525→   525→   525→                const arduinoResult = await this.compilerService.tryArduinoBuild(buildDir, sketchName, fqbn);
   526→   526→   526→                if (arduinoResult && arduinoResult.success && arduinoResult.binaryPath) {
   527→   527→   527→                    binaryPath = arduinoResult.binaryPath;
   528→   528→   528→                } else {
   529→   529→   529→                    const arduinoAvailable = this.compilerService.findArduinoCli() !== undefined;
   530→   530→   530→                    if (arduinoAvailable) {
   531→   531→   531→                        return {
   532→   532→   532→                            success: false,
   533→   533→   533→                            output: compileResult.output,
   534→   534→   534→                            error: 'Firmware binary (.bin) not found. Arduino CLI build failed. ' +
   535→   535→   535→                                'Check the compiler output for errors. Ensure ESP32 board support is installed: ' +
   536→   536→   536→                                'arduino-cli core install esp32:esp32',
   537→   537→   537→                        };
   538→   538→   538→                    }
   539→   539→   539→                    return {
   540→   540→   540→                        success: false,
   541→   541→   541→                        output: compileResult.output,
   542→   542→   542→                        error: 'Firmware binary (.bin) not found. Install Arduino CLI for full compilation: ' +
   543→   543→   543→                            'https://arduino.github.io/arduino-cli/latest/',
   544→   544→   544→                    };
   545→   545→   545→                }
   546→   546→   546→            }
   547→   547→   547→        }
   548→   548→   548→
   549→   549→   549→        // ── Step 3: Flash the firmware ────────────────────────────────
   550→   550→   550→        const flashRequest: FlashRequest = {
   551→   551→   551→            binaryPath,
   552→   552→   552→            chipType,
   553→   553→   553→            baudRate: 460800,
   554→   554→   554→        };
   555→   555→   555→
   556→   556→   556→        if (portPath) {
   557→   557→   557→            flashRequest.portPath = portPath;
   558→   558→   558→        }
   559→   559→   559→
   560→   560→   560→        return this.flash(flashRequest);
   561→   561→   561→    }

   652→   652→   652→    /**
   653→   653→   653→     * Run the flash subprocess and parse its output for progress.
   654→   654→   654→     */
   655→   655→   655→    private executeFlash(
   656→   656→   656→        cmd: string,
   657→   657→   657→        args: string[],
   658→   658→   658→        portUsed: string,
   659→   659→   659→        onProgress?: ProgressCallback,
   660→   660→   660→    ): Promise<FlashResult> {
   661→   661→   661→        return new Promise(resolve => {
   662→   662→   662→            const proc = spawn(cmd, args, { stdio: 'pipe' });
   663→   663→   663→
   664→   664→   664→            let stdout = '';
   665→   665→   665→            let stderr = '';
   666→   666→   666→            let resolved = false;
   667→   667→   667→
   668→   668→   668→            const finish = (result: FlashResult) => {
   669→   669→   669→                if (resolved) {
   670→   670→   670→                    return;
   671→   671→   671→                }
   672→   672→   672→                resolved = true;
   673→   673→   673→                resolve(result);
   674→   674→   674→            };
   675→   675→   675→
   676→   676→   676→            proc.stdout.on('data', (data: Buffer) => {
   677→   677→   677→                const text = data.toString();
   678→   678→   678→                stdout += text;
   679→   679→   679→                this.parseProgress(text, onProgress);
   680→   680→   680→            });
   681→   681→   681→
   682→   682→   682→            proc.stderr.on('data', (data: Buffer) => {
   683→   683→   683→                const text = data.toString();
   684→   684→   684→                stderr += text;
   685→   685→   685→                // esptool writes most of its output to stderr
   686→   686→   686→                this.parseProgress(text, onProgress);
   687→   687→   687→            });
   688→   688→   688→
   689→   689→   689→            proc.on('close', (code: number | null) => {
   690→   690→   690→                const fullOutput = stdout + '\n' + stderr;
   691→   691→   691→
   692→   692→   692→                if (code === 0) {
   693→   693→   693→                    finish({
   694→   694→   694→                        success: true,
   695→   695→   695→                        output: fullOutput.trim(),
   696→   696→   696→                        portUsed,
   697→   697→   697→                    });
   698→   698→   698→                } else {
   699→   699→   699→                    const error = this.classifyFlashError(stderr || stdout, code);
   700→   700→   700→                    finish({
   701→   701→   701→                        success: false,
   702→   702→   702→                        output: fullOutput.trim(),
   703→   703→   703→                        error,
   704→   704→   704→                        portUsed,
   705→   705→   705→                    });
   706→   706→   706→                }
   707→   707→   707→            });
   708→   708→   708→
   709→   709→   709→            proc.on('error', (err: Error) => {
   710→   710→   710→                if (err.message.includes('ENOENT')) {
   711→   711→   711→                    finish({
   712→   712→   712→                        success: false,
   713→   713→   713→                        output: '',
   714→   714→   714→                        error: `esptool command not found: ${cmd}. Please install esptool (pip install esptool).`,
   715→   715→   715→                        portUsed,
   716→   716→   716→                    });
   717→   717→   717→                } else {
   718→   718→   718→                    finish({
   719→   719→   719→                        success: false,
   720→   720→   720→                        output: '',
   721→   721→   721→                        error: `Failed to start esptool: ${err.message}`,
   722→   722→   722→                        portUsed,
   723→   723→   723→                    });
   724→   724→   724→                }
   725→   725→   725→            });
   726→   726→   726→
   727→   727→   727→            // Timeout
   728→   728→   728→            setTimeout(() => {
   729→   729→   729→                if (!resolved) {
   730→   730→   730→                    proc.kill();
   731→   731→   731→                    finish({
   732→   732→   732→                        success: false,
   733→   733→   733→                        output: (stdout + '\n' + stderr).trim(),
   734→   734→   734→                        error: `Flash operation timed out after ${FLASH_TIMEOUT_MS / 1000} seconds. ` +
   735→   735→   735→                               'The board may not be responding. Try pressing the BOOT button while flashing.',
   736→   736→   736→                        portUsed,
   737→   737→   737→                    });
   738→   738→   738→                }
   739→   739→   739→            }, FLASH_TIMEOUT_MS);
   740→   740→   740→        });
   741→   741→   741→    }
   742→   742→   742→
   743→   743→   743→    /**
   744→   744→   744→     * Parse esptool output lines for progress information.
   745→   745→   745→     *
   746→   746→   746→     * Recognised patterns:
   747→   747→   747→     *  - "Writing at 0x... (N%)"   → writing progress
   748→   748→   748→     *  - "Hash of data verified"    → 100 % verification step
   749→   749→   749→     *  - "A fatal error occurred"   → fatal failure
   750→   750→   750→     *  - "Chip is ESP32..."         → chip detection confirmation
   751→   751→   751→     */
   752→   752→   752→    private parseProgress(text: string, onProgress?: ProgressCallback): void {
   753→   753→   753→        if (!onProgress) {
   754→   754→   754→            return;
   755→   755→   755→        }
   756→   756→   756→
   757→   757→   757→        // Split on newlines — esptool often uses \r for progress updates
   758→   758→   758→        const lines = text.split(/\r?\n|\r/);
   759→   759→   759→
   760→   760→   760→        for (const line of lines) {
   761→   761→   761→            const trimmed = line.trim();
   762→   762→   762→            if (!trimmed) {
   763→   763→   763→                continue;
   764→   764→   764→            }
   765→   765→   765→
   766→   766→   766→            // Writing progress: "Writing at 0x00010000... (12%)"
   767→   767→   767→            const writeMatch = trimmed.match(/Writing at 0x[0-9a-fA-F]+\s*\((\d+)%\)/);
   768→   768→   768→            if (writeMatch) {
   769→   769→   769→                const percent = parseInt(writeMatch[1], 10);
   770→   770→   770→                // Writing is roughly 0‑80 % of the total operation
   771→   771→   771→                const scaled = Math.round(percent * 0.8);
   772→   772→   772→                onProgress(scaled, `Writing firmware... ${percent}%`);
   773→   773→   773→                continue;
   774→   774→   774→            }
   775→   775→   775→
   776→   776→   776→            // Verification: "Hash of data verified."
   777→   777→   777→            if (trimmed.includes('Hash of data verified')) {
   778→   778→   778→                onProgress(100, 'Firmware verified successfully.');
   779→   779→   779→                continue;
   780→   780→   780→            }
   781→   781→   781→
   782→   782→   782→            // Chip detection: "Chip is ESP32-D0WDQ6 (revision v1.0)"
   783→   783→   783→            const chipMatch = trimmed.match(/Chip is (ESP[^\(]+)/i);
   784→   784→   784→            if (chipMatch) {
   785→   785→   785→                onProgress(5, `Detected chip: ${chipMatch[1].trim()}`);
   786→   786→   786→                continue;
   787→   787→   787→            }
   788→   788→   788→
   789→   789→   789→            // Connecting: "Connecting...."
   790→   790→   790→            if (trimmed.includes('Connecting')) {
   791→   791→   791→                onProgress(2, 'Connecting to board...');
   792→   792→   792→                continue;
   793→   793→   793→            }
   794→   794→   794→
   795→   795→   795→            // Erasing flash
   796→   796→   796→            if (trimmed.toLowerCase().includes('erasing') || trimmed.toLowerCase().includes('erase')) {
   797→   797→   797→                onProgress(10, 'Erasing flash...');
   798→   798→   798→                continue;
   799→   799→   799→            }
   800→   800→   800→
   801→   801→   801→            // Compressed size info
   802→   802→   802→            const compMatch = trimmed.match(/Wrote (\d+) bytes.*compressed/);
   803→   803→   803→            if (compMatch) {
   804→   804→   804→                onProgress(85, `Wrote ${compMatch[1]} bytes (compressed), verifying...`);
   805→   805→   805→                continue;
   806→   806→   806→            }
   807→   807→   807→
   808→   808→   808→            // Leaving...
   809→   809→   809→            if (trimmed.includes('Leaving...')) {
   810→   810→   810→                onProgress(95, 'Finishing up...');
   811→   811→   811→                continue;
   812→   812→   812→            }
   813→   813→   813→
   814→   814→   814→            // Fatal error
   815→   815→   815→            if (trimmed.includes('A fatal error occurred')) {
   816→   816→   816→                onProgress(-1, `Error: ${trimmed}`);
   817→   817→   817→                continue;
   818→   818→   818→            }
   819→   819→   819→        }

   822→   822→   822→    /**
   823→   823→   823→     * Classify a flash failure into a user-friendly error message.
   824→   824→   824→     */
   825→   825→   825→    private classifyFlashError(output: string, exitCode: number | null): string {
   826→   826→   826→        if (!output) {
   827→   827→   827→            return `esptool exited with code ${exitCode} (no output).`;
   828→   828→   828→        }
   829→   829→   829→
   830→   830→   830→        const lower = output.toLowerCase();
   831→   831→   831→
   832→   832→   832→        // Board not found / no serial data
   833→   833→   833→        if (lower.includes('failed to connect') || lower.includes('no serial data received')) {
   834→   834→   834→            return (
   835→   835→   835→                'Could not connect to the ESP32 board. Please ensure:\n' +
   836→   836→   836→                '  • The board is connected via USB.\n' +
   837→   837→   837→                '  • The correct port is selected.\n' +
   838→   838→   838→                '  • No other program (serial monitor, IDE) is using the port.\n' +
   839→   839→   839→                '  • Try pressing and holding the BOOT button while flashing.'
   840→   840→   840→            );
   841→   841→   841→        }
   842→   842→   842→
   843→   843→   843→        // Port access denied
   844→   844→   844→        if (lower.includes('permission denied') || lower.includes('access is denied') || lower.includes('could not open port')) {
   845→   845→   845→            return (
   846→   846→   846→                'Serial port access denied. You may need to:\n' +
   847→   847→   847→                '  • Add your user to the dialout group (Linux): sudo usermod -aG dialout $USER\n' +
   848→   848→   848→                '  • Close other programs using the port.\n' +
   849→   849→   849→                '  • Run the IDE as administrator (Windows).'
   850→   850→   850→            );
   851→   851→   851→        }
   852→   852→   852→
   853→   853→   853→        // Chip not in bootloader mode
   854→   854→   854→        if (lower.includes('wrong boot mode') || lower.includes('download mode')) {
   855→   855→   855→            return (
   856→   856→   856→                'The chip is not in download mode. Try:\n' +
   857→   857→   857→                '  • Hold the BOOT button, then click Upload, and release BOOT after "Connecting..." appears.\n' +
   858→   858→   858→                '  • Check that EN/RST and BOOT pins are correctly wired.'
   859→   859→   859→            );
   860→   860→   860→        }
   861→   861→   861→
   862→   862→   862→        // Flash verification failure
   863→   863→   863→        if (lower.includes('verify failed') || lower.includes('md5') && lower.includes('mismatch')) {
   864→   864→   864→            return (
   865→   865→   865→                'Flash verification failed. The firmware was written but the read-back did not match.\n' +
   866→   866→   866→                'Possible causes: bad USB cable, unstable power supply, or defective flash chip.'
   867→   867→   867→            );
   868→   868→   868→        }
   869→   869→   869→
   870→   870→   870→        // Generic fatal error from esptool
   871→   871→   871→        const fatalMatch = output.match(/A fatal error occurred:\s*(.*)/i);
   872→   872→   872→        if (fatalMatch) {
   873→   873→   873→            return `esptool error: ${fatalMatch[1].trim()}`;
   874→   874→   874→        }
   875→   875→   875→
   876→   876→   876→        // Timed out
   877→   877→   877→        if (lower.includes('timeout') || lower.includes('timed out')) {
   878→   878→   878→            return 'Connection timed out. The board may not be responding. Try pressing the BOOT button while flashing.';
   879→   879→   879→        }
   880→   880→   880→
   881→   881→   881→        // Fallback — return the last few lines of output
   882→   882→   882→        const lastLines = output.split('\n').filter(l => l.trim()).slice(-3).join('\n');
   883→   883→   883→        return `Flash failed (exit code ${exitCode}):\n${lastLines}`;
   884→   884→   884→    }

   886→   886→   886→    /**
   887→   887→   887→     * Normalise a chip type string to one of the supported values.
   888→   888→   888→     * Accepts case-insensitive input with or without hyphens/spaces.
   889→   889→   889→     */
   890→   890→   890→    private normalizeChipType(raw: string): string | undefined {
   891→   891→   891→        const normalized = raw.toLowerCase().replace(/[\s\-_]/g, '');
   892→   892→   892→        const knownTypes = ['esp32', 'esp32s2', 'esp32s3', 'esp32c3', 'esp8266'];
   893→   893→   893→

   895→   895→   895→        if (knownTypes.includes(normalized)) {
   896→   896→   896→            return normalized;
   897→   897→   897→        }
   898→   898→   898→
   899→   899→   899→        // Handle "esp32-s2" → "esp32s2" (already handled by the replace above)
   900→   900→   900→        // Handle just "s2" / "s3" / "c3"
   901→   901→   901→        if (['s2', 's3', 'c3'].includes(normalized)) {
   902→   902→   902→            return `esp32${normalized}`;
   903→   903→   903→        }
   904→   904→   904→
   905→   905→   905→        return undefined;
   906→   906→   906→    }
